# The sound engine SDK for the PC (docs/engines.md): every engine as its own library, a generic
# contract test suite that runs on each of them, the renderer and the host shell. It needs only
# dsp/ (the engines), tools/harness/ (the scripted performances and the key-stream replay) and
# third_party/doctest: no firmware, no ChibiOS. Built as part of the firmware repository's host
# presets, or on its own:  cmake -S sdk -B build/sdk && cmake --build build/sdk && ctest --test-dir build/sdk
#
#   striso_engine_<name>       one engine (dsp/include/striso_dsp/engines/<name>.h): its EngineEntry,
#                              striso_engine_<name>()
#   striso_effect_fx           the master effects: striso::sdk::fx_effect()
#   striso_engines             every engine, by name: striso::sdk::engines(), find_engine()
#   striso_performances        the scripted performances of tools/harness, by name
#   striso_sdk                 the MIDI/MPE mapping, the MIDI file reader, the players, the shell
#   striso_render              the renderer (sdk/tools/striso_render.cpp)
#   render_custom              the renderer with one engine and performance compiled with extra
#                              flags (STRISO_RENDER_*: tools/harness/render_strisy.sh)
#   tests: contract.<name>.*   the contract suite on each engine (tests/contract.cpp)
#          sdk.*               the shared helpers, the MIDI mapping and files, the shell
#          striso_render.*     the renderer's log names its engine, preset and input; bad inputs fail
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.25)
  project(striso_sdk DESCRIPTION "Striso sound engine SDK" LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../dsp" "${CMAKE_BINARY_DIR}/dsp")
  add_library(striso_host_options INTERFACE)
  target_compile_options(striso_host_options INTERFACE -Wall -Wextra)
  enable_testing()
endif()
set(STRISO_HARNESS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../tools/harness" CACHE PATH "tools/harness: the scripted performances, harness_util.h, rec/")
set(STRISO_DOCTEST_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/doctest" CACHE PATH "doctest")
cmake_path(NORMAL_PATH STRISO_HARNESS_DIR)
cmake_path(NORMAL_PATH STRISO_DOCTEST_DIR)
set(_gen "${CMAKE_CURRENT_BINARY_DIR}/generated")

# every engine and effect library: the flags of the renderers (-O2, no FMA contraction: renders are
# reproducible and the golden checksums hold)
add_library(striso_sdk_options INTERFACE)
target_compile_options(striso_sdk_options INTERFACE -O2 -ffp-contract=off)
target_include_directories(striso_sdk_options INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(striso_sdk_options INTERFACE striso::dsp striso_host_options)

  target_link_libraries(striso_engine_${v} PUBLIC striso_sdk_options)
  string(APPEND _entries "const striso::sdk::EngineEntry& striso_engine_${v}();\n")

add_library(striso_effect_fx STATIC "${_gen}/effect_fx.cpp")
target_link_libraries(striso_effect_fx PUBLIC striso_sdk_options)

# --- the scripted performances (tools/harness/performance.h, perf_<name>.h) ----------------------------
function(striso_performance_source out name header)
  string(MAKE_C_IDENTIFIER "${name}" id)
  file(CONFIGURE OUTPUT "${out}" @ONLY CONTENT "// generated by sdk/CMakeLists.txt: the scripted performance \"${name}\" (${header})
#include <math.h>
#include <stdint.h>
#include <stdio.h>
#include <string.h>

#include \"striso_sdk/performance.h\"

namespace perf_${id} {
#include \"${header}\"
}

static void perf_block(int b, int nv, striso::dsp::VoiceControls* vc, striso::dsp::GlobalControls& g) {
	perf_${id}::Ctl v[NVOICES_MAX];
	float gl[8];
	perf_${id}::block_controls(b, v, nv, gl);
	for (int i = 0; i < nv; i++) {
		vc[i].note = v[i].note;
		vc[i].pres = v[i].pres;
		vc[i].vpres = v[i].vpres;
		vc[i].but_x = v[i].but_x;
		vc[i].but_y = v[i].but_y;
	}
	g = {gl[0], gl[1], gl[2], gl[3], gl[4], gl[5], gl[6], gl[7]};
}

extern const striso::sdk::ScriptedPerformance striso_perf_${id};
const striso::sdk::ScriptedPerformance striso_perf_${id} = {\"${name}\", \"${header}\", SAMPLERATE, NVOICES_MAX, perf_block};
")
endfunction()

file(GLOB _perf_headers "${STRISO_HARNESS_DIR}/perf_*.h")
list(SORT _perf_headers)
list(PREPEND _perf_headers "${STRISO_HARNESS_DIR}/performance.h")
set(_perf_srcs "")
set(_perf_decls "")
set(_perf_list "")
set(STRISO_PERFORMANCES "")
foreach(h IN LISTS _perf_headers)
  get_filename_component(hn "${h}" NAME_WE)
  if(hn STREQUAL "performance")
    set(name default)
  else()
    string(REGEX REPLACE "^perf_" "" name "${hn}")
  endif()
  string(MAKE_C_IDENTIFIER "${name}" id)
  striso_performance_source("${_gen}/perf_${id}.cpp" "${name}" "${h}")
  list(APPEND _perf_srcs "${_gen}/perf_${id}.cpp")
  list(APPEND STRISO_PERFORMANCES ${name})
  string(APPEND _perf_decls "extern const striso::sdk::ScriptedPerformance striso_perf_${id};\n")
  string(APPEND _perf_list "\t    &striso_perf_${id},\n")
endforeach()
set(STRISO_PERFORMANCES "${STRISO_PERFORMANCES}" PARENT_SCOPE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${STRISO_HARNESS_DIR}")
file(CONFIGURE OUTPUT "${_gen}/performances.cpp" @ONLY CONTENT "// generated by sdk/CMakeLists.txt: the scripted performances of tools/harness
#include \"striso_sdk/performance.h\"

${_perf_decls}
std::span<const striso::sdk::ScriptedPerformance* const> striso::sdk::performances() {
	static const ScriptedPerformance* const list[] = {
${_perf_list}	};
	return list;
}
")
add_library(striso_performances STATIC ${_perf_srcs} "${_gen}/performances.cpp")
target_link_libraries(striso_performances PUBLIC striso_sdk_options)
target_compile_definitions(striso_performances PRIVATE _USE_MATH_DEFINES)
# (the performances are 1990s C: unused helpers, old-style casts)
target_compile_options(striso_performances PRIVATE -Wno-unused-function -Wno-unused-variable -Wno-unused-parameter)

# --- the SDK: MIDI in, MIDI files, the players, the shell ---------------------------------------------
add_library(striso_sdk STATIC src/midi_in.cpp src/smf.cpp src/shell.cpp src/performance.cpp)
target_include_directories(striso_sdk PRIVATE "${STRISO_HARNESS_DIR}")
target_link_libraries(striso_sdk PUBLIC striso_sdk_options striso_effect_fx)

add_executable(striso_render tools/striso_render.cpp)
target_link_libraries(striso_render PRIVATE striso_sdk striso_engines striso_performances)

# --- render_custom: one engine and one performance with extra compiler flags ---------------------------
# (tools/harness/render_strisy.sh configures these). The performance is compiled into the engine's
# translation unit, so its test hooks (BOWED_TEST_HOOK and the like) reach the engine.
set(STRISO_RENDER_VOICE "strisy" CACHE STRING "render_custom: engine")
set(STRISO_RENDER_PERF "performance.h" CACHE STRING "render_custom: performance header")
set(STRISO_RENDER_PERF_DIR "" CACHE PATH "render_custom: extra directory searched for the performance header first")
set(STRISO_RENDER_FLAGS "" CACHE STRING "render_custom: extra compiler flags")
if(STRISO_RENDER_VOICE IN_LIST STRISO_VOICES)
  separate_arguments(_custom_flags UNIX_COMMAND "${STRISO_RENDER_FLAGS}")
  set(v "${STRISO_RENDER_VOICE}")
  # the performance: in STRISO_RENDER_PERF_DIR if it is there, else in tools/harness (the log and
  # the WAV name the file used)
  set(_perf "${STRISO_HARNESS_DIR}/${STRISO_RENDER_PERF}")
  if(STRISO_RENDER_PERF_DIR AND EXISTS "${STRISO_RENDER_PERF_DIR}/${STRISO_RENDER_PERF}")
    set(_perf "${STRISO_RENDER_PERF_DIR}/${STRISO_RENDER_PERF}")
  elseif(IS_ABSOLUTE "${STRISO_RENDER_PERF}")
    set(_perf "${STRISO_RENDER_PERF}")
  endif()
  if(NOT EXISTS "${_perf}")
    message(FATAL_ERROR "render_custom: no performance ${STRISO_RENDER_PERF} (in ${STRISO_RENDER_PERF_DIR} or ${STRISO_HARNESS_DIR})")
  endif()
  file(CONFIGURE OUTPUT "${_gen}/custom.cpp" @ONLY CONTENT "// generated by sdk/CMakeLists.txt: render_custom's engine (${v}) and performance (${STRISO_RENDER_PERF})
#include \"striso_dsp/engines/${v}.h\"
#include \"engine_entry.h\"
#include \"striso_sdk/performance.h\"
#include <math.h>
#include <stdint.h>
#include <stdio.h>
#include <string.h>
#include \"${_perf}\"

static void perf_block(int b, int nv, striso::dsp::VoiceControls* vc, striso::dsp::GlobalControls& g) {
	Ctl v[NVOICES_MAX];
	float gl[8];
	block_controls(b, v, nv, gl);
	for (int i = 0; i < nv; i++) {
		vc[i].note = v[i].note;
		vc[i].pres = v[i].pres;
		vc[i].vpres = v[i].vpres;
		vc[i].but_x = v[i].but_x;
		vc[i].but_y = v[i].but_y;
	}
	g = {gl[0], gl[1], gl[2], gl[3], gl[4], gl[5], gl[6], gl[7]};
}
static const striso::sdk::ScriptedPerformance perf = {\"custom\", \"${_perf}\", SAMPLERATE, NVOICES_MAX, perf_block};

std::span<const striso::sdk::EngineEntry* const> striso::sdk::engines() {
	static const EngineEntry* const list[] = {&entry_of<striso::dsp::engines::${v}>(\"${STRISO_VOICE_${v}_HEADER}\")};
	return list;
}
std::span<const striso::sdk::ScriptedPerformance* const> striso::sdk::performances() {
	static const ScriptedPerformance* const list[] = {&perf};
	return list;
}
")
  add_executable(render_custom tools/striso_render.cpp "${_gen}/custom.cpp" src/registry.cpp)
  target_include_directories(render_custom PRIVATE "${STRISO_HARNESS_DIR}")
  target_link_libraries(render_custom PRIVATE striso_sdk)
  target_compile_definitions(render_custom PRIVATE _USE_MATH_DEFINES)
  target_compile_options(render_custom PRIVATE ${_custom_flags})
else()
  message(WARNING "STRISO_RENDER_VOICE=${STRISO_RENDER_VOICE} is not an engine (${STRISO_VOICES})")
endif()

# --- tests ------------------------------------------------------------------------------------------
if(NOT EXISTS "${STRISO_DOCTEST_DIR}/doctest/doctest.h")
  message(WARNING "no doctest at ${STRISO_DOCTEST_DIR}: the SDK's tests are not built")
  return()
endif()
include("${STRISO_DOCTEST_DIR}/cmake/doctest.cmake")

# the contract suite on every engine: one test program per engine, linked to that engine's library
foreach(v IN LISTS STRISO_VOICES)
  add_executable(striso_engine_${v}_contract tests/main.cpp tests/contract.cpp)
  target_compile_definitions(striso_engine_${v}_contract PRIVATE STRISO_ENGINE_ENTRY=striso_engine_${v})
  target_include_directories(striso_engine_${v}_contract PRIVATE "${STRISO_DOCTEST_DIR}")
  target_link_libraries(striso_engine_${v}_contract PRIVATE striso_engine_${v} striso_sdk_options)
  doctest_discover_tests(striso_engine_${v}_contract TEST_PREFIX "contract.${v}." PROPERTIES LABELS "contract;unit")

# the renderer says what it played, and refuses what it can't find
set(_rec "${STRISO_HARNESS_DIR}/rec/play_2026-10-22_staccato.csv")
add_test(NAME striso_render.log_names_engine_preset_performance
  COMMAND striso_render --engine va --preset pad --seconds 0.5)
set_tests_properties(striso_render.log_names_engine_preset_performance PROPERTIES LABELS unit
  PASS_REGULAR_EXPRESSION "engine va \\(voice_va.h\\), preset [0-9]+ \"pad\".*input scripted performance \"default\" \\([^\n]*performance.h \\([0-9]+ bytes, fnv [0-9a-f]+\\)\\) \\[the default input")
if(EXISTS "${_rec}")
  add_test(NAME striso_render.log_names_recording COMMAND striso_render --engine ep --rec "${_rec}" --seconds 0.5)
  set_tests_properties(striso_render.log_names_recording PROPERTIES LABELS unit
    PASS_REGULAR_EXPRESSION "engine ep \\(voice_ep.h\\), preset \\(its default\\) 0 .*input key recording [^\n]*play_2026-10-22_staccato.csv \\([0-9]+ bytes, fnv [0-9a-f]+\\): [0-9]+ key events")
endif()
foreach(t IN ITEMS "unknown_engine;--engine nosuch" "unknown_preset;--engine va --preset nosuch"
                   "unknown_performance;--engine va --perf nosuch" "missing_recording;--engine va --rec /nonexistent.csv"
                   "missing_midi;--engine va --midi /nonexistent.mid" "two_inputs;--engine va --perf default --rec x.csv"
                   "too_many_voices;--engine trumpet --nv 64")
  list(GET t 0 name)
  list(GET t 1 args)
  separate_arguments(args)
  add_test(NAME striso_render.refuses_${name} COMMAND striso_render ${args})
  set_tests_properties(striso_render.refuses_${name} PROPERTIES LABELS unit WILL_FAIL TRUE)
endforeach()
