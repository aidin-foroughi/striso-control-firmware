# Host tests (ctest --preset host, or host-asan under ASan + UBSan).
#
#   unit.*      doctest cases (striso_tests): live-parameter fuzz of every engine and the effects,
#               the tuning math of the StcN command (every key's pitch matches its name),
#               the bowed engine's seed bounds, the parameter tables, the config record format
#               (config_store.c), the DSP headers compiling on their own
#   usb_audio_core   tools/usbaudio/test_core.c: the USB audio ring and clock servo simulation
#   striso_py   tools/striso.py's UF2 and config record handling (no board needed)
#   render.<voice>[.p<preset>].<performance>   every engine and preset renders the default
#               performance, the stale-vpres performance and the recorded key streams
#               (tools/harness/rec): finite, peak <= 8, a silent idle tail after the stale
#               release, and the golden checksum (tests/golden/renders.txt) so any change of the
#               DSP's output is caught. After an intended change: tools/update_goldens.sh

# --- doctest unit tests ----------------------------------------------------------------------------
# striso_engines.h: every voice engine, and STRISO_TEST_ENGINES(X, XP): X(ns) for an engine,
# XP(ns) for an engine with presets (from tools/harness: STRISO_VOICES)
set(_eng "// generated by tests/CMakeLists.txt\n#pragma once\n#include \"striso_dsp/strisy.h\"\n")
set(_list "")
foreach(v IN LISTS STRISO_VOICES)
  if(NOT v STREQUAL "strisy")
    string(APPEND _eng "#include \"striso_dsp/voice_${v}.h\"\n")
  endif()
  if(STRISO_VOICE_${v}_PRESETS)
    string(APPEND _list " XP(${v})")
  else()
    string(APPEND _list " X(${v})")
  endif()
endforeach()
string(APPEND _eng "#define STRISO_TEST_ENGINES(X, XP)${_list}\n")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/striso_engines.h" CONTENT "${_eng}")

set(_golden_file "${CMAKE_CURRENT_SOURCE_DIR}/golden/renders.txt")

file(GLOB _recordings "${PROJECT_SOURCE_DIR}/tools/harness/rec/*.csv")
list(SORT _recordings)

# a released voice's stale positive vpres must not keep it sounding (synth_control leaves it):
# silent and idle 12 s after the release at 1 s (the longest preset release, va pad, ends ~9.5 s)
set(STRISO_STALE_SILENT_AFTER 12)

function(_render_test name exe)
  set(args --nv 10 --check --golden-name ${name} ${ARGN})
  if(STRISO_GOLDEN)
    if(DEFINED "_golden_${name}")
      list(APPEND args --expect "${_golden_${name}}")
    else()
      list(APPEND args --expect missing)
    endif()
  endif()
  add_test(NAME ${name} COMMAND ${exe} ${args})
  set_tests_properties(${name} PROPERTIES LABELS render)
endfunction()

foreach(v IN LISTS STRISO_VOICES)
  set(npresets "${STRISO_VOICE_${v}_PRESETS}")
  if(npresets)  # (not "n": if(n) is the constant N, false)
    math(EXPR last "${npresets} - 1")
    set(presets "")
    foreach(p RANGE ${last})
      list(APPEND presets ${p})
    endforeach()
  else()
    set(presets none)
  endif()
  foreach(p IN LISTS presets)
    if(p STREQUAL "none")
      set(tag "${v}")
      set(parg "")
    else()
      set(tag "${v}.p${p}")
      set(parg --preset ${p})
    endif()
    _render_test(render.${tag}.default render_${v} ${parg})
    _render_test(render.${tag}.stale render_${v}_stale ${parg} --silent-after ${STRISO_STALE_SILENT_AFTER})
    foreach(rec IN LISTS _recordings)
      get_filename_component(rn "${rec}" NAME_WE)
      string(REGEX REPLACE "^play_" "" rn "${rn}")
      _render_test(render.${tag}.rec_${rn} render_${v} ${parg} --rec "${rec}")
    endforeach()
  endforeach()
  if(TARGET render_${v}_perf)
    _render_test(render.${v}.perf render_${v}_perf)
  endif()
endforeach()
# master effects on the default performance (stereo)
_render_test(render.strisy.fx render_strisy --fx)
_render_test(render.va.p5.fx render_va --preset 5 --fx)
